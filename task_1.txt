create database movie_stream_db;

use movie_stream_db;

create table movies(movie_id int auto_increment primary key,
                    movie_name varchar(20) not null, 
                    genre enum("action","romance","thriller"),
                    language varchar(50),
                    rating decimal(2,1),
                    duration_min int,
                    release_year int);

#18. Find average rating.

#19. Count movies in each genre (GROUP BY).

select genre,count(*) as movie_count from movies group by genre;

#20. Find average rating for each language.

select language,avg(rating) as average_rating from movies group by language;

#21. Display genres having more than 1 movie (HAVING).

select genre,count(*) as movie_count from movies group by genre having count(*) > 1;

#22. Display languages having more than 1 movie (HAVING).

select language,count(*) as movie_count from movies group by language having count(*) > 1;
